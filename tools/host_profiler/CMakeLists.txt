cmake_minimum_required(VERSION 3.20)
project(shinka_host_profiler LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The host stack sampler requires 64-bit Windows")
endif()
add_executable(profile_host_stack ../profile_host_stack.cpp)
target_compile_features(profile_host_stack PRIVATE cxx_std_17)
target_link_libraries(profile_host_stack PRIVATE dbghelp)
if(MSVC)
    target_compile_options(profile_host_stack PRIVATE /W4 /EHsc)
endif()
